def git(lane, *args):
    return subprocess.run(["git", "-C", lane] + list(args), capture_output=True, check=True).stdout


def rows(allow):
    n = 0
    if allow and os.path.exists(allow):
        with open(allow, encoding="utf-8") as fh:
            for line in fh:
                line = line.rstrip("\n")
                if not line or line.startswith("#"):
                    continue
                cells = line.split("\t")
                if len(cells) == 4 and cells[1] == "aws-sts-session-token":
                    raise OSError("allowlist row")
                n += 1
    return n
