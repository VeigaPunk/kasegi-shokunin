import importlib.util, sys, hashlib
spec = importlib.util.spec_from_file_location("scan", "corpus/instruments/capture-secret-scan.py")
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
pats = m.patterns()
GH = b"ghr_" + b"aB3"*25 + b"a"   # 76
def probe(name, data):
    found, allowed = m.scan_bytes(data, pats)
    got = ",".join(sorted({s for s,_ in found})) or "none"
    spans = []
    for s, rx in pats:
        for mm in rx.finditer(data):
            spans.append((s, mm.start(), mm.end()))
    print("%-42s -> %-24s spans=%s" % (name, got, spans if len(spans)<4 else len(spans)))
    return got

# --- FP probes ---
probe("FP1 word-before ghr_76", b"x"+GH+b" end")

probe("FP3 ghr_76 hex-looking", b"ghr_" + b"0123456789abcdef"*4 + b"0123456789ab")
probe("FP4 ghr_76 + hyphen suffix", GH + b"-tail")
probe("FP5 ghr_76 inside base64ish blob", b"QUJD"+GH+b"REVG")
probe("FP6 glcbt prose <header>", b"glcbt-eyJ<header>.eyJ<payload>.<signature>")
probe("FP7 S140 regex source line", b"regex = '''\\b(?:[a-zA-Z0-9]{1,20}-)?glcbt-eyJ[0-9a-zA-Z_\\-]{10,1000}\\.eyJ[0-9a-zA-Z_\\-]{10,1000}\\.[0-9a-zA-Z_\\-]{20,1000}'''")
probe("FP8 S140 redacted-line text", b"[line redacted: a published rule example value from GitLab's secret-detection rules file; raw snapshot digest and the redacted line list are in this receipt]")
probe("FP9 glcbt + generic eyJ jwt prose", b"the token looks like glcbt-eyJ<redacted> x")
