# the evaluator, in full, so the verdict is re-derivable
#!/usr/bin/env python3
"""Score a registry fixture against the frozen round-40 C2 kill rule (claims.json)."""
import json
import pathlib
import subprocess
import sys

NINE = ["S016", "S017", "S019", "S020", "S021", "S022", "S023", "S024", "S025"]
IDX = {"id": 0, "teaches": 6, "status": 7, "used_by": 10}


def parse(path):
    rows = {}
    total = 0
    for line in pathlib.Path(path).read_text().splitlines():
        if not line.startswith("|"):
            continue
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        if len(cells) < 11 or not cells[0].startswith("S"):
            continue
        total += 1
        rows[cells[0]] = {k: cells[i] for k, i in IDX.items()}
    return rows, total


def evidence(row, freeze):
    """Gate-block evidence for one decided row; returns (ok, detail)."""
    if row["status"] == "rejected":
        return ("flip:" in row["teaches"].lower(),
                "reason+flip" if "flip:" in row["teaches"].lower() else "reason-only,no-flip")
    if row["status"] == "mined":
        entry = row["used_by"]
        if entry in ("", "\u2014", "-"):
            return False, "no-entry"
        md = freeze / "corpus" / entry
        if not md.is_file() or f"source: {row['id']}" not in md.read_text():
            return False, "entry-missing-or-unbound"
        return False, "entry-bound,no-capture-manifest-in-fixture"
    return False, "still-queued"


def main():
    reg = pathlib.Path(sys.argv[1])
    freeze = pathlib.Path(sys.argv[2])
    claims_path = pathlib.Path(sys.argv[3])
    claims = json.loads(claims_path.read_text())
    c2 = [c for c in claims["claims"] if c["id"] == "C2"][0]
    rows, total = parse(reg)
    nine = {sid: rows.get(sid, {"status": "absent", "teaches": "", "used_by": ""}) for sid in NINE}

    queued = sum(1 for r in nine.values() if r["status"] == "queued")
    mined = sum(1 for r in nine.values() if r["status"] == "mined")
    absent = sum(1 for r in nine.values() if r["status"] == "absent")
    details = {sid: evidence(r, freeze) for sid, r in nine.items()}
    ok_evid = sum(1 for ok, _ in details.values() if ok)
    s060 = sorted(sid for sid in rows if len(sid) == 4 and sid.startswith("S0") and int(sid[1:]) >= 60)
    s060_entries = sum(
        1 for p in (freeze / "corpus" / "knowledge").glob("*.md")
        if any(l.startswith("source: S06") for l in p.read_text().splitlines()))
    frontier = subprocess.run(
        ["grep", "-cE", r"^\| F[0-9]+ \|", str(freeze / "corpus" / "README.md")],
        capture_output=True, text=True)
    frontier_n = int(frontier.stdout.strip() or 0)
    harness = subprocess.run([sys.executable, str(freeze / "corpus" / "instruments" / "frontier-recheck-count.py")],
                             cwd=str(freeze), capture_output=True, text=True)
    hsum = [l for l in harness.stdout.splitlines() if l.startswith("entries=")]
    hsum = hsum[0] if hsum else "<no-summary>"

    print(f"frozen claims: {claims_path} (C2 kill_rule sha256-locked at claims freeze)")
    print("frozen C2 kill_rule:")
    print("  " + c2["kill_rule"])
    print()
    print(f"registry fixture: {reg} rows={total} nine_absent={absent}")
    print()
    fired = []

    def clause(n, name, observed, required, ok):
        verdict = "pass" if ok else "FIRED"
        if not ok:
            fired.append(f"{n:02d} {name}")
        print(f"clause-{n:02d} {name:34s} observed={observed!s:12s} required={required!s:14s} {verdict}")

    clause(1, "nine-rows-still-queued", queued, "0", queued == 0)
    clause(2, "mined-among-nine-at-least-4", mined, ">=4", mined >= 4)
    clause(3, "decided-row-gate-evidence", f"{ok_evid}/9", "9/9", ok_evid == 9)
    clause(4, "s060-row-opened-this-round", len(s060), ">=1", len(s060) >= 1)
    clause(5, "knowledge-entry-citing-s060", s060_entries, ">=1", s060_entries >= 1)
    clause(6, "frontier-rows", frontier_n, "23", frontier_n == 23)
    clause(7, "harness-entries26-pass26", hsum, "entries=26 pass=26 exit=0", hsum == "entries=26 pass=26 exit=0")
    print(f"clause-08 source-recheck-fails-for-a-mined-row  observed=n/a (mined among nine = {mined}) "
          f"{'pass' if mined == 0 else 'FIRED'}")
    if mined:
        fired.append("08 mined-row-without-capture-manifest")
    print("clause-09 census-exit-or-new-W/E      observed=not-evaluated (fixture is a detached copy; not a census root)")
    print()
    print("per-row Gate-block evidence:")
    for sid in NINE:
        ok, detail = details[sid]
        print(f"  {sid} status={nine[sid]['status']:9s} evidence={'yes' if ok else 'NO '} ({detail})")
    print()
    verdict = "REJECTED" if fired else "ACCEPTED"
    print(f"clause-10 junk-twin-accepted                 observed={verdict:12s} required=not-accepted "
          f"{'pass' if verdict == 'REJECTED' else 'FIRED'}")
    print(f"fired={len(fired)} " + ",".join(fired))
    print(f"verdict={verdict}")
    return 1 if fired else 0


if __name__ == "__main__":
    sys.exit(main())

## H. Reading

The control fails on independent clauses, so it cannot be rescued by satisfying one:
- clause 2 (mined among the nine >= 4): observed 0, because a rejected status cannot be mined;
- clause 3 (every decided row carries its Gate-block evidence): 0/9, because `no time` and
  `superseded` are reasons with no flip, no capture, no entry and no passing source-recheck;
- clauses 4-7: the do-nothing baseline opens no S060+ row, adds no S060+ entry, leaves the
  frontier at 22 and the harness at 25/25.
The adversarial variant shows the counter is not defeated by the cheap repair either: setting
all nine to `mined` clears the >=4 floor but the entry-count equality and the missing capture
manifests still reject it.

Positive controls F show the same probes return nonzero on landed evidence (6/6 rejected rows
carry flip; S029 is bound to its entry; S057's capture passes source-recheck), so the zeros in
C and D are measurements of the control, not a dead detector.

exit-code note: `grep -c` and `grep -l ... | wc -l` report exit=1/0 for zero hits as shown;
`python3 eval-c2.py` exits 1 exactly when at least one clause fires.
