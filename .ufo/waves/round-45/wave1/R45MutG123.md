#!/usr/bin/env python3
"""Mutation recipe for corpus/instruments/r45-c2-gate.py gates G1-G3 (scratch only).

Builds a hand-made tree in which G1-G3 are satisfied (base export + F27 row +
S071 registry row + F27 entry + copied S071 snapshot/manifest), then applies one
single-edit mutant at a time to a fresh copy and runs the counter (read-only).
rejected=1 iff the target gate prints pass=0 for that mutant.
"""
import os
import re
import shutil
import subprocess
import sys

SCRATCH = os.environ["MUT_SCRATCH"]
TMPDIR = os.path.join(SCRATCH, "tmp")
R = "/home/vgpnk/Projects/shokunin/kasegi-shokunin"

ENV = dict(os.environ, TMPDIR=TMPDIR)

F27_ROW = ("| F27 | Linux kernel security-bugs reporting process | admitted | "

S071_ROW = ("| S071 | Linux kernel security bugs reporting process — " + URL + " | docs | A | "
            "The Linux kernel's own security-bugs reporting process (embargo ceiling, AI-assisted reports, binary-only refusal, reproducibility bar) | "
            "sha256 0918dcfa7979f348c9004bf3ce6e183507367185fe69f4e0c7688521e9346f40, capture corpus/source-snapshots/round-45/S071.txt | "
            "fixture paraphrase, not a quote | mined | 2026-10-28 | 180d | " + ENTRY + " |")
ENTRY_TEXT = (

    "source: S071 — " + URL + " — capture corpus/source-snapshots/round-45/S071.txt\n"

ENTRY_FULL = ENTRY_TEXT + (

    "Spans (fixture: verbatim from the pinned capture):\n"

def line_index(lines, prefix):
    idx = [i for i, ln in enumerate(lines) if ln.startswith(prefix)]
    assert len(idx) == 1, (prefix, idx)
    return idx[0]


def read(p):
    with open(p, encoding="utf-8") as fh:
        return fh.read()


def write(p, t):
    with open(p, "w", encoding="utf-8") as fh:
        fh.write(t)


def build(dst, full=False):
    shutil.copytree(PURE, dst, symlinks=True)
    d = os.path.join(dst, "corpus/source-snapshots/round-45")
    os.makedirs(d, exist_ok=True)
    shutil.copy(os.path.join(SNAP, "S071.txt"), d)
    shutil.copy(os.path.join(SNAP, "manifest.json"), d)
    write(os.path.join(dst, ENTRY), ENTRY_FULL if full else ENTRY_TEXT)
    rp = os.path.join(dst, "corpus/README.md")
    lines = read(rp).split("\n")
    i = line_index(lines, "| F26 |")
    lines.insert(i + 1, F27_ROW)
    write(rp, "\n".join(lines))
    sp = os.path.join(dst, "corpus/SOURCES.md")
    lines = read(sp).split("\n")
    i = line_index(lines, "| S070 |")
    lines.insert(i + 1, S071_ROW)
    write(sp, "\n".join(lines))


def edit_line(rel, prefix, fn):
    """Return a mutator that rewrites exactly one line starting with prefix."""
    def mut(dst):
        p = os.path.join(dst, rel)
        lines = read(p).split("\n")
        i = line_index(lines, prefix)
        old = lines[i]
        new = fn(old)
        assert new != old
        lines[i] = new
        write(p, "\n".join(lines))
        return old, new
    return mut


def insert_after(rel, prefix, new_lines):
    def mut(dst):
        p = os.path.join(dst, rel)
        lines = read(p).split("\n")
        i = line_index(lines, prefix)
        old = lines[i]
        lines[i + 1:i + 1] = new_lines
        write(p, "\n".join(lines))
        return old, "\n".join(new_lines)
    return mut


def wrap_in_comment(rel, prefix):
    def mut(dst):
        p = os.path.join(dst, rel)
        lines = read(p).split("\n")
        i = line_index(lines, prefix)
        old = lines[i]
        lines[i:i + 1] = ["<!--", old, "-->"]
        write(p, "\n".join(lines))
        return old, "<!--\\n" + old + "\\n-->"
    return mut


README = "corpus/README.md"
SOURCES = "corpus/SOURCES.md"
ENTRY_REL = ENTRY

MUTANTS = [
    # name, target gate, description, mutator
    ("g1-id-gap", "G1",
     "README: F05 row id renamed to F28 (row count and admitted count unchanged)",
     edit_line(README, "| F05 |", lambda s: s.replace("| F05 |", "| F28 |", 1))),
    ("g1-status-demoted", "G1",
     "README: F27 status admitted -> candidate",
     edit_line(README, "| F27 |", lambda s: s.replace("| admitted |", "| candidate |", 1))),
    ("g1-row-in-html-comment", "G1",
     "README: F27 row wrapped in an HTML comment (<!-- ... -->, two added lines)",
     wrap_in_comment(README, "| F27 |")),
    ("g2-recheck-placeholder", "G2",
     "entry header: recheck line replaced by a placeholder that is only longer than 10 chars",
     edit_line(ENTRY_REL, "recheck:", lambda s: "recheck: TBD-to-be-run-later")),
    ("g2-source-url-suffix", "G2",
     "entry header: source URL suffixed with -archive (startswith check)",
     edit_line(ENTRY_REL, "source: S071", lambda s: s.replace(URL, URL + "-archive", 1))),
    ("g2-verified-date-off", "G2",
     "entry header: verified-at 2026-10-28 -> 2026-10-27 (control: must be rejected)",
     edit_line(ENTRY_REL, "verified-at:", lambda s: s.replace("2026-10-28", "2026-10-27", 1))),
    ("g3-malformed-dup-row", "G3",
     "SOURCES: a 10-cell duplicate S071 row inserted after the valid S071 row",
     insert_after(SOURCES, "| S071 |", ["| S071 | stale duplicate | docs | C | x | x | x | rejected | 2026-10-01 | 180d |"])),
    ("g3-tier-flipped", "G3",
     "SOURCES: S071 tier A -> B (control: must be rejected)",
     edit_line(SOURCES, "| S071 |", lambda s: s.replace("| docs | A |", "| docs | B |", 1))),
]


def run_counter(root):
    p = subprocess.run([sys.executable, COUNTER, "--root", root], capture_output=True, text=True, env=ENV, timeout=600)
    gates = {}
    for ln in p.stdout.splitlines():
        m = re.match(r"^gate (G\d) pass=(\d) read=(.*)$", ln)
        if m:
            gates[m.group(1)] = (int(m.group(2)), m.group(3))
    total = [ln for ln in p.stdout.splitlines() if ln.startswith("c2-gate ")]
    return gates, total, p.returncode, p.stdout, p.stderr


def probe(name, dst):
    """Prints the mutant's defect-present evidence straight from the mutated bytes."""
    rd = lambda rel: read(os.path.join(dst, rel)).split("\n")
    if name == "g1-id-gap":
        ids = [re.match(r"^\| (F\d+) \|", ln).group(1) for ln in rd(README) if re.match(r"^\| F\d+ \|", ln)]
        return f"frontier-ids={len(ids)} has-F05={'F05' in ids} has-F28={'F28' in ids} F01..F27-missing={sorted(set(f'F{i:02d}' for i in range(1, 28)) - set(ids))}"
    if name == "g1-row-in-html-comment":
        L = rd(README); i = [k for k, ln in enumerate(L) if ln.startswith("| F27 |")][0]
        return f"line-before-F27={L[i-1]!r} line-after-F27={L[i+1]!r}"
    if name == "g1-status-demoted":
        return "F27-row-status=" + [ln.split("|")[3].strip() for ln in rd(README) if ln.startswith("| F27 |")][0]
    if name.startswith("g2-recheck"):
        return "recheck-line=" + [ln for ln in rd(ENTRY) if ln.startswith("recheck:")][0]
    if name.startswith("g2-source"):
        return "source-line=" + [ln for ln in rd(ENTRY) if ln.startswith("source:")][0]
    if name.startswith("g2-verified"):
        return "verified-line=" + [ln for ln in rd(ENTRY) if ln.startswith("verified-at:")][0]
    if name == "g3-malformed-dup-row":
        rows = [ln for ln in rd(SOURCES) if re.match(r"^ {0,3}\|\s*S071\s*\|", ln)]
        return "S071-rows=" + " ; ".join(f"cells={len([x for x in r.strip().strip('|').split('|')])}" for r in rows)
    if name == "g3-tier-flipped":
        return "S071-tier-cell=" + [r.strip().strip("|").split("|")[3].strip() for r in rd(SOURCES) if re.match(r"^ {0,3}\|\s*S071\s*\|", r)][0]
    return "none"


def run_mutant(kind, src_tree, name, target, desc, mut, mutdir):
    dst = os.path.join(mutdir, kind + "-" + name)
    if os.path.exists(dst):
        shutil.rmtree(dst)
    shutil.copytree(src_tree, dst, symlinks=True)
    old, new = mut(dst)
    pr = probe(name, dst)
    gates, total, rc, out, err = run_counter(dst)
    pass_target, read_target = gates[target]
    rejected = 0 if pass_target == 1 else 1
    print(f"tree={kind} mutant={name} rejected={rejected}")
    print(f"  target={target} pass={pass_target} read={read_target}")
    print(f"  desc: {desc}")
    print(f"  defect-probe: {pr}")
    print(f"  edit-old: {old!r}")
    print(f"  edit-new: {new!r}")
    print("  gates: " + " | ".join(f"{g} pass={gates[g][0]}" for g in ("G1", "G2", "G3", "G4", "G5", "G6", "G7")))
    print(f"  {total[0] if total else '?'} rc={rc}")
    shutil.rmtree(dst)
    return name, rejected


def main():
    os.makedirs(TMPDIR, exist_ok=True)
    base = os.path.join(SCRATCH, "base")
    built = os.path.join(SCRATCH, "built")
    built_full = os.path.join(SCRATCH, "built-full")
    mutdir = os.path.join(SCRATCH, "mut")
    for d in (base, built, built_full, mutdir):
        shutil.rmtree(d, ignore_errors=True)
    shutil.copytree(PURE, base, symlinks=True)
    build(built)
    build(built_full, full=True)

    print("== controls ==")
    for label, root in (("do-nothing-base", base), ("built-unmutated", built), ("built-full-unmutated", built_full)):
        gates, total, rc, out, err = run_counter(root)
        for g in ("G1", "G2", "G3"):
            print(f"control={label} gate={g} pass={gates[g][0]} read={gates[g][1]}")
        print(f"control={label} {total[0] if total else '?'} rc={rc}")
        print(f"control={label} all-gates=" + " ".join(f"{g}:{gates[g][0]}" for g in sorted(gates)))
        if err.strip():
            print("stderr:", err.strip()[:400])
    assert all(run_counter(built)[0][g][0] == 1 for g in ("G1", "G2", "G3")), "hand-built tree does not satisfy G1-G3"
    assert all(run_counter(built_full)[0][g][0] == 1 for g in ("G1", "G2", "G3", "G4", "G5", "G6", "G7")), "full tree not 7/7"

    for kind, src_tree in (("core", built), ("full", built_full)):
        print(f"== mutants on tree={kind} ==")
        results = [run_mutant(kind, src_tree, *m, mutdir) for m in MUTANTS]
        surv = [n for n, r in results if r == 0]
        print(f"tree={kind} survivors=" + (",".join(surv) if surv else "none"))
        print(f"tree={kind} total-mutants={len(results)} rejected={sum(r for _, r in results)} survived={len(surv)}")


if __name__ == "__main__":
    main()
