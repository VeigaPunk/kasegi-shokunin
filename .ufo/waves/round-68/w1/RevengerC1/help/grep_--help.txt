Search for PATTERN in each FILE or standard input.

Usage: grep [OPTIONS] [ARGS]...

Arguments:
  [ARGS]...  PATTERN followed by FILEs (PATTERN is omitted with -e or -f)

Options:
  -e, --regexp <PATTERN>          Use PATTERN for matching (may be repeated; all patterns are OR-ed)
  -f, --file <FILE>               Read patterns from FILE, one per line
  -E, --extended-regexp           Interpret PATTERN as a strict extended regular expression
  -G, --basic-regexp              Interpret PATTERN using the default basic-compatible mode
  -F, --fixed-strings             Interpret PATTERN as a fixed string
  -P, --perl-regexp               Interpret PATTERN as a Perl-compatible regular expression
  -i, --ignore-case               Ignore case distinctions in patterns and data
      --no-ignore-case            Restore case-sensitive matching after an earlier -i
  -v, --invert-match              Select non-matching lines
  -w, --word-regexp               Match only whole words
  -x, --line-regexp               Match only whole lines
  -c, --count                     Print only a count of selected lines per FILE
  -l, --files-with-matches        Print only the names of FILEs with at least one selected line
  -L, --files-without-match       Print only the names of FILEs with no selected lines
  -m, --max-count <NUM>           Stop after NUM selected lines in each input
  -o, --only-matching             Print only the matched non-empty parts of selected lines
  -q, --quiet                     Quiet; suppress normal output and stop after the first selected line [alias: --silent]
  -s, --no-messages               Suppress error messages about nonexistent or unreadable files
  -b, --byte-offset               Prefix output with the zero-based byte offset
  -H, --with-filename             Always print the file name with output lines
  -h, --no-filename               Never print the file name with output lines
      --label <LABEL>             Use LABEL as the displayed name for standard input
  -n, --line-number               Prefix each output line with its one-based line number
  -T, --initial-tab               Align line content on a tab stop after output prefixes
  -Z, --null                      Write NUL instead of the separator following a file name
  -A, --after-context <NUM>       Print NUM lines of trailing context after selected lines
  -B, --before-context <NUM>      Print NUM lines of leading context before selected lines
  -C, --context <NUM>             Print NUM lines of leading and trailing context
      --group-separator <STRING>  Print STRING between non-adjacent groups of context lines
      --no-group-separator        Do not print a separator between context groups
  -a, --text                      Process binary input as text
  -I                              Treat binary input as having no selected lines
      --binary-files <TYPE>       Choose how binary input is searched [possible values: binary, text, without-match]
  -D, --devices <ACTION>          Choose how device, FIFO, and socket operands are handled [possible values: read, skip]
  -d, --directories <ACTION>      Choose how directory operands are handled [possible values: read, skip, recurse]
      --include <GLOB>            Search files matching GLOB
      --exclude <GLOB>            Skip files matching GLOB
      --exclude-from <FILE>       Read file exclusion globs from FILE
      --exclude-dir <GLOB>        Skip directories matching GLOB during recursive searches
      --include-dir <GLOB>        Search directories matching GLOB during recursive searches
  -r, --recursive                 Recursively search each directory operand
  -R, --dereference-recursive     Recursively search and follow every symbolic link
  -O                              Follow symbolic links named as command-line operands
  -p                              Do not follow symbolic links during recursive searches
  -S                              Follow every symbolic link during recursive searches
      --line-buffered             Flush standard output after each output record
  -U, --binary                    Use binary I/O where the platform distinguishes it
  -z, --null-data                 Treat NUL rather than newline as the input and output record delimiter
      --mmap                      Request memory-mapped input where supported
  -u                              Accepted compatibility option with no effect
      --help                      Print a help message
  -V, --version                   Print version information
      --color[=<WHEN>]            Accept color configuration without injecting ANSI into redirected output
