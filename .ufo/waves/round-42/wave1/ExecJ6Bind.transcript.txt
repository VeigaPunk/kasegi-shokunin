  #!/usr/bin/env python3
  """Apply one F-BIND mutation to a scratch copy of the lane corpus.
  
  Usage: mutate.py <variant> <lane-copy-root>
  
  The mutated record is always round-41's S063 (the record the round-41 wave-2
  attack used). Every variant writes exactly one change and prints it.
  """
  import json
  import os
  import sys
  
  VARIANTS = ("a-missing-path", "b-mutated-url", "c-mutated-sha256",
              "d-url-trailing-slash", "e-capture-url-trailing-slash",
              "f-mutated-snapshot-sha256")
  
  
  def main(argv):
      if len(argv) != 3 or argv[1] not in VARIANTS:
          print(__doc__, file=sys.stderr)
          return 2
      variant, root = argv[1], argv[2]
      snap = os.path.join(root, "corpus", "source-snapshots", "round-41")
      manifest_path = os.path.join(snap, "manifest.json")
      with open(manifest_path, encoding="utf-8") as fh:
          manifest = json.load(fh)
      record = [item for item in manifest["sources"] if item.get("id") == "S063"][0]
  
      if variant == "a-missing-path":
          target = os.path.join(snap, record["path"])
          os.remove(target)
          print("mutation=%s removed=%s" % (variant, target))
          print("record path still names %r" % record["path"])
          return 0
      if variant == "b-mutated-url":
          before = record["url"]
          record["url"] = before.replace("/rfc/rfc9116.txt", "/RFC/rfc9116.txt")
          print("mutation=%s url before=%s after=%s" % (variant, before, record["url"]))
      elif variant == "c-mutated-sha256":
          before = record["sha256"]
          record["sha256"] = "f" * 64 if before != "f" * 64 else "e" * 64
          print("mutation=%s sha256 before=%s after=%s" % (variant, before, record["sha256"]))
      elif variant == "d-url-trailing-slash":
          before = record["url"]
          record["url"] = before + "/"
          print("mutation=%s url before=%s after=%s" % (variant, before, record["url"]))
      elif variant == "e-capture-url-trailing-slash":
          before = record["capture_url"]
          record["capture_url"] = before + "/"
          print("mutation=%s capture_url before=%s after=%s final_url=%s"
                % (variant, before, record["capture_url"], record["http"]["final_url"]))
      elif variant == "f-mutated-snapshot-sha256":
          before = record["snapshot_sha256"]
          record["snapshot_sha256"] = "f" * 64 if before != "f" * 64 else "e" * 64
          print("mutation=%s snapshot_sha256 before=%s after=%s (sha256 left untouched)"
                % (variant, before, record["snapshot_sha256"]))
      with open(manifest_path, "w", encoding="utf-8") as fh:
          json.dump(manifest, fh, indent=2, sort_keys=True)
          fh.write("\n")
      return 0
  
  
  if __name__ == "__main__":
      sys.exit(main(sys.argv))
